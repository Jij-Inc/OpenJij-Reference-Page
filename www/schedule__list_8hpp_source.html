166ClassicalScheduleList
167make_classical_schedule_list(double beta_min, double beta_max,
168 std::size_t one_mc_step,
169 std::size_t num_call_updater) {
170 auto schedule_list = ClassicalScheduleList(num_call_updater);
171 const double ratio = beta_min == beta_max ? 1.0 : beta_max / beta_min;
172 if (num_call_updater > 1 && beta_min > 0.0 && beta_max > 0.0 &&
173 std::isfinite(beta_min) && std::isfinite(beta_max) &&
174 !std::isnormal(ratio)) {
175 // A ratio can overflow or lose precision even with finite endpoints.
176 const std::size_t last = num_call_updater - 1;
177 const double log_min = std::log(beta_min);
178 const double log_span = std::log(beta_max) - log_min;
179 for (std::size_t index = 0; index < num_call_updater; ++index) {
180 const double fraction =
181 static_cast<double>(index) / static_cast<double>(last);
182 auto &schedule = schedule_list[index];
183 schedule.one_mc_step = one_mc_step;
184 schedule.updater_parameter =
185 ClassicalUpdaterParameter(std::exp(log_min + fraction * log_span));
186 }
187 schedule_list.front().updater_parameter.beta = beta_min;
188 schedule_list.back().updater_parameter.beta = beta_max;
189 return schedule_list;
190 }
191
192 const double r_beta =
193 std::pow(ratio, 1.0 / static_cast<double>(num_call_updater - 1));
194 const auto [lower, upper] = std::minmax(beta_min, beta_max);
195 double beta = beta_min;
196 for (auto &schedule : schedule_list) {
197 schedule.one_mc_step = one_mc_step;
198 schedule.updater_parameter = ClassicalUpdaterParameter(beta);
199 // Rounding in the step ratio can carry the accumulated beta past an endpoint.
200 beta = std::clamp(beta * r_beta, lower, upper);
201 }
202 if (num_call_updater > 1) {
203 schedule_list.back().updater_parameter.beta = beta_max;
204 }
205
206 return schedule_list;
207}

219TransverseFieldScheduleList
220make_transverse_field_schedule_list(double beta, std::size_t one_mc_step,
221 std::size_t num_call_updater) {
222 double ds = 1.0 / static_cast<double>(num_call_updater - 1);
223 double s = 0;
224
225 auto schedule_list = TransverseFieldScheduleList(num_call_updater);
226 for (auto &schedule : schedule_list) {
227 schedule.one_mc_step = one_mc_step;
228 schedule.updater_parameter = TransverseFieldUpdaterParameter(beta, s);
229 s += ds;
230 }
231
232 return schedule_list;
233}

247ClassicalConstraintScheduleList make_classical_constraint_schedule_list(
248 double lambda, double beta_min, double beta_max, std::size_t one_mc_step,
249 std::size_t num_call_updater) {
250 double r_beta = std::pow(beta_max / beta_min,
251 1.0 / static_cast<double>(num_call_updater - 1));
252 double beta = beta_min;
253
254 auto schedule_list = ClassicalConstraintScheduleList(num_call_updater);
255 for (auto &schedule : schedule_list) {
256 schedule.one_mc_step = one_mc_step;
257 schedule.updater_parameter =
258 ClassicalConstraintUpdaterParameter(beta, lambda);
259 beta *= r_beta;
260 }
261
262 return schedule_list;
263}

301template<typename FloatType>
302std::vector<FloatType> GenerateLinearBetaSchedule(const FloatType beta_min, const FloatType beta_max, const std::int32_t num_sweeps) {
303 if (num_sweeps == 1) {
304 return std::vector<FloatType>{beta_min};
305 }
306 std::vector<FloatType> beta_list(num_sweeps);
307 for (std::int32_t i = 0; i < num_sweeps; ++i) {
308 beta_list[i] = beta_min + i*(beta_max - beta_min)/(num_sweeps - 1);
309 }
310 return beta_list;
311}

319template<typename FloatType>
320std::vector<FloatType> GenerateGeometricBetaSchedule(const FloatType beta_min, const FloatType beta_max, const std::int32_t num_sweeps) {
321 if (num_sweeps == 1) {
322 return std::vector<FloatType>{beta_min};
323 }
324 std::vector<FloatType> beta_list(num_sweeps);
325 const FloatType alpha = std::pow(beta_max/beta_min, 1/static_cast<FloatType>(num_sweeps - 1));
326 FloatType beta = beta_min;
327 for (std::int32_t i = 0; i < num_sweeps; ++i) {
328 beta_list[i] = beta;
329 beta = beta*alpha;
330 }
331 return beta_list;
332}

341template<typename FloatType>
342std::vector<FloatType> GenerateBetaList(const TemperatureSchedule schedule_type,
343 const FloatType beta_min,
344 const FloatType beta_max,
345 const std::int32_t num_sweeps) {
346 std::vector<FloatType> beta_list;
347 if (schedule_type == TemperatureSchedule::LINEAR) {
348 beta_list = GenerateLinearBetaSchedule(beta_min, beta_max, num_sweeps);
349 }
350 else if (schedule_type == TemperatureSchedule::GEOMETRIC) {
351 beta_list = GenerateGeometricBetaSchedule(beta_min, beta_max, num_sweeps);
352 }
353 else {
354 throw std::runtime_error("Unknwon beta schedule list");
355 }
356 return beta_list;
357}
